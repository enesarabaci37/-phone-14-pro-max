'=====================================================================
'  SAYI / BİRİM YARDIMCILARI (clsOlay da kullanır)
'=====================================================================
Public Function SayiOku(ByVal s As String) As Double
    s = Replace(Trim$(s), ",", ".")
    SayiOku = Val(s)
End Function

Public Function SayiYaz(ByVal v As Double) As String
    Dim s As String
    s = Format(Round(v, 4), "0.####")
    If VBA.Right$(s, 1) = "," Or VBA.Right$(s, 1) = "." Then s = VBA.Left$(s, Len(s) - 1)
    SayiYaz = s
End Function

Public Function AdimBuyuklugu() As Double
    Select Case cmbBirim.ListIndex
        Case 1: AdimBuyuklugu = 0.1
        Case 2: AdimBuyuklugu = 0.01
        Case Else: AdimBuyuklugu = 1
    End Select
End Function

Private Function BirimCarpani(ByVal idx As Long) As Double
    Select Case idx
        Case 1: BirimCarpani = 10
        Case 2: BirimCarpani = 1000
        Case Else: BirimCarpani = 1
    End Select
End Function

Private Function BirimAdi() As String
    Select Case cmbBirim.ListIndex
        Case 1: BirimAdi = "cm"
        Case 2: BirimAdi = "m"
        Case Else: BirimAdi = "mm"
    End Select
End Function

Private Function OlcuYazi(ByVal mm As Double) As String
    OlcuYazi = SayiYaz(mm / birimCarpan)
End Function

Private Sub BirimEtiketleriniGuncelle()
    Dim lb As Variant
    For Each lb In birimEtiketleri
        lb.Caption = BirimAdi()
    Next lb
End Sub

Private Sub cmbBirim_Change()
    Dim yeni As Double, eski As Boolean
    yeni = BirimCarpani(cmbBirim.ListIndex)
    If Not yukleniyor Then
        eski = yukleniyor
        yukleniyor = True
        BirimCevir txtGen, yeni
        BirimCevir txtYuk, yeni
        BirimCevir txtKenar, yeni
        BirimCevir txtBosluk, yeni
        BirimCevir txtPlakaAra, yeni
        yukleniyor = eski
    End If
    birimCarpan = yeni
    BirimEtiketleriniGuncelle
    AyarDegisti
End Sub

Private Sub BirimCevir(tb As MSForms.TextBox, ByVal yeni As Double)
    tb.Text = SayiYaz(SayiOku(tb.Text) * birimCarpan / yeni)
End Sub

'=====================================================================
'  AYAR DEĞİŞİNCE
'=====================================================================
Public Sub AyarDegisti()
    If yukleniyor Then Exit Sub
    AlanDurumlari
    If chkCanli.Value Then
        Hesapla
        gosterPlaka = 1
        Ciz
    Else
        lblBilgi.Caption = "Ayarlar değişti. Önizlemeyi yenilemek için 'Önizle' düğmesine basın."
    End If
End Sub

Private Sub AlanDurumlari()
    Dim eski As Boolean, w As Double, h As Double
    eski = yukleniyor
    txtArama.Enabled = (cmbKalite.ListIndex = 2)
    If chkSayfa.Value Then
        If SayfaBoyutu(w, h) Then
            yukleniyor = True
            txtGen.Text = OlcuYazi(w)
            txtYuk.Text = OlcuYazi(h)
            yukleniyor = eski
        End If
        txtGen.Enabled = False: txtYuk.Enabled = False: btnTakas.Enabled = False
    Else
        txtGen.Enabled = True: txtYuk.Enabled = True: btnTakas.Enabled = True
    End If
    txtPlakaAra.Enabled = (cmbPlakaYer.ListIndex <> 2)
End Sub

Private Function SayfaBoyutu(ByRef w As Double, ByRef h As Double) As Boolean
    Dim doc As Object, eskiBirim As Long
    On Error GoTo Yok
    Set doc = AktifBelge()
    If doc Is Nothing Then Exit Function
    eskiBirim = doc.Unit
    doc.Unit = cdrMillimeter
    w = doc.ActivePage.SizeWidth
    h = doc.ActivePage.SizeHeight
    doc.Unit = eskiBirim
    SayfaBoyutu = True
    Exit Function
Yok:
    SayfaBoyutu = False
End Function

Private Sub OnAyarListesiDoldur()
    Dim v As Variant, i As Long, eski As Boolean
    If GetSetting(UYG, "OnAyar", VARSAYILAN_AD, "") = "" Then
        SaveSetting UYG, "OnAyar", VARSAYILAN_AD, VARSAYILAN_AYAR
    End If
    eski = yukleniyor
    yukleniyor = True
    cmbOnAyar.Clear
    v = GetAllSettings(UYG, "OnAyar")
    If IsArray(v) Then
        For i = LBound(v, 1) To UBound(v, 1)
            cmbOnAyar.AddItem CStr(v(i, 0))
        Next i
    End If
    yukleniyor = eski
End Sub

Private Sub OnAyarSec(ByVal ad As String)
    Dim i As Long, eski As Boolean
    eski = yukleniyor
    yukleniyor = True
    For i = 0 To cmbOnAyar.ListCount - 1
        If cmbOnAyar.List(i) = ad Then cmbOnAyar.ListIndex = i: Exit For
    Next i
    yukleniyor = eski
End Sub

Private Sub cmbOnAyar_Change()
    Dim s As String
    If yukleniyor Then Exit Sub

Private Function MmMetin(ByVal ekranDegeri As String) As String
    MmMetin = Trim$(Str$(Round(SayiOku(ekranDegeri) * birimCarpan, 4)))
End Function

Private Function Deger(ByVal s As String, ByVal anahtar As String, ByVal vars As Double) As Double
    Dim p As Variant, kv() As String
    Deger = vars
    For Each p In Split(s, ";")
        kv = Split(CStr(p), "=")
        If UBound(kv) = 1 Then
            If LCase$(Trim$(kv(0))) = anahtar Then Deger = Val(kv(1)): Exit Function
        End If
    Next p
End Function

'=====================================================================
'  HESAP (CorelDRAW'daki nesnelere dokunmaz)
'=====================================================================
Private Sub Hesapla()
    Dim doc As Object, sr As Object, eskiBirim As Long
    Dim m As Long, n As Long, i As Long, c As Long, k As Long, adet As Long
    Dim bx As Double, byy As Double, bw As Double, bh As Double
    Dim w0() As Double, h0() As Double, aw() As Double, ah() As Double
    Dim matW As Double, matH As Double, usW As Double, usH As Double
    Dim allowRot As Boolean, kalite As Long, arama As Long
    Dim sh As Shape

    hesapOK = False
    rSheets = 0
    nItem = 0
    On Error GoTo Hata

    adim = "belge"
    Set doc = AktifBelge()
    If doc Is Nothing Then
        hesapMesaj = "Açık bir CorelDRAW belgesi yok."
        Exit Sub
    End If
    adim = "seçim"
    Set sr = SecimAl(doc)
    m = SecimDiziyeAl(sr)
    If m = 0 Then
        hesapMesaj = "Dizilecek nesneleri CorelDRAW'da seçin," & vbCrLf & _
                     "sonra 'Önizle / Seçimi yenile' düğmesine basın." & vbCrLf & vbCrLf & _
                     "(Bu pencere açıkken seçim yapabilirsiniz.)"
        Exit Sub
    End If

    ' --- Parametreler (mm)
    adim = "ayarlar"
    If chkSayfa.Value Then
        If Not SayfaBoyutu(matW, matH) Then hesapMesaj = "Sayfa boyutu okunamadı.": Exit Sub
    Else
        matW = SayiOku(txtGen.Text) * birimCarpan
        matH = SayiOku(txtYuk.Text) * birimCarpan
    End If
    pKenar = SayiOku(txtKenar.Text) * birimCarpan
    pGap = SayiOku(txtBosluk.Text) * birimCarpan
    adet = CLng(SayiOku(txtAdet.Text))
    allowRot = (cmbDondur.ListIndex = 1)
    kalite = cmbKalite.ListIndex
    arama = CLng(SayiOku(txtArama.Text))
    If arama < 1 Then arama = 1
    If arama > 500 Then arama = 500
    pBasla = cmbBasla.ListIndex
    pKontur = CBool(chkKontur.Value)

    If matW <= 0 Or matH <= 0 Then hesapMesaj = "Malzeme genişliği ve yüksekliği 0'dan büyük olmalı.": Exit Sub
    If pKenar < 0 Or pGap < 0 Then hesapMesaj = "Boşluk ve kenar payı negatif olamaz.": Exit Sub
    If adet < 1 Then hesapMesaj = "Adet en az 1 olmalı.": Exit Sub
    usW = matW - 2 * pKenar
    usH = matH - 2 * pKenar
    If usW <= 0 Or usH <= 0 Then hesapMesaj = "Kenar payı malzemeye göre çok büyük.": Exit Sub
    If m * adet > 5000 Then hesapMesaj = "En fazla 5000 nesne dizilebilir (şu an " & m * adet & ").": Exit Sub

    ' --- Nesne ölçüleri
    adim = "nesne ölçüleri"
    ReDim w0(1 To m): ReDim h0(1 To m)
    eskiBirim = doc.Unit
    doc.Unit = cdrMillimeter
    For i = 1 To m
        adim = "nesne ölçüleri " & i & "/" & m
        Set sh = secS(i)
        sh.GetBoundingBox bx, byy, bw, bh, pKontur
        w0(i) = bw: h0(i) = bh
    Next i
    doc.Unit = eskiBirim

    For i = 1 To m
        If Not ((w0(i) <= usW + EPS And h0(i) <= usH + EPS) Or _
                (allowRot And h0(i) <= usW + EPS And w0(i) <= usH + EPS)) Then
            hesapMesaj = "Bir nesne malzemeye sığmıyor:" & vbCrLf & _
                         OlcuYazi(w0(i)) & " x " & OlcuYazi(h0(i)) & " " & BirimAdi() & vbCrLf & _
                         "Kullanılabilir alan: " & OlcuYazi(usW) & " x " & OlcuYazi(usH) & " " & BirimAdi() & _
                         IIf(allowRot, "", vbCrLf & "(Döndürmeye izin vermeyi deneyin.)")
            Exit Sub
        End If
    Next i

    ' --- Parça listesi
    n = m * adet
    ReDim iSrc(1 To n): ReDim iKop(1 To n): ReDim iW(1 To n): ReDim iH(1 To n)
    ReDim aw(1 To n): ReDim ah(1 To n)
    k = 0
    For i = 1 To m
        For c = 1 To adet
            k = k + 1
            iSrc(k) = i: iKop(k) = c
            iW(k) = w0(i): iH(k) = h0(i)
            aw(k) = w0(i) + pGap: ah(k) = h0(i) + pGap
        Next c
    Next i
    nItem = n
    pMatW = matW: pMatH = matH
    pSW = usW + pGap: pSH = usH + pGap

    adim = "yerleşim hesabı"
    lblBilgi.Caption = "Hesaplanıyor..."
    DoEvents
    rSheets = EnIyiYerlesim(n, aw, ah, pSW, pSH, allowRot, kalite, arama)
    If rSheets <= 0 Then
        hesapMesaj = "Yerleşim bulunamadı."
        Exit Sub
    End If
    hesapOK = True
    Exit Sub

Hata:
    hesapMesaj = "Hesaplama hatası (" & adim & "): " & Err.Description & " [" & Err.Number & "]"
    hesapOK = False
End Sub

'=====================================================================
'  ÖNİZLEME ÇİZİMİ
'=====================================================================
Private Sub Ciz()
    Dim PW As Double, PH As Double, sc As Double, sx As Double, sy As Double
    Dim k As Long, c As Long, adet As Long
    Dim x As Double, y As Double, w As Double, h As Double, dolu As Double, topDolu As Double
    Dim lb As MSForms.Label

    frOn.Controls.Clear

    If Not hesapOK Then
        Set lb = frOn.Controls.Add("Forms.Label.1")
        With lb
            .Left = 20: .Top = 140: .Width = frOn.InsideWidth - 40: .Height = 100
            .Caption = hesapMesaj
            .TextAlign = fmTextAlignCenter
            .Font.Size = 10
            .ForeColor = RGB(120, 120, 120)
        End With
        lblPlaka.Caption = ""
        lblBilgi.Caption = ""
        btnOnceki.Enabled = False
        btnSonraki.Enabled = False
        Exit Sub
    End If

    If gosterPlaka < 1 Then gosterPlaka = 1
    If gosterPlaka > rSheets Then gosterPlaka = rSheets

    PW = frOn.InsideWidth
    PH = frOn.InsideHeight
    sc = Min2((PW - 24) / pMatW, (PH - 24) / pMatH)
    sx = (PW - pMatW * sc) / 2
    sy = (PH - pMatH * sc) / 2

    ' Plaka
    Set lb = frOn.Controls.Add("Forms.Label.1")
    With lb
        .Left = sx: .Top = sy: .Width = Max2(pMatW * sc, 2): .Height = Max2(pMatH * sc, 2)
        .BackColor = RGB(255, 255, 255)
        .BorderStyle = fmBorderStyleSingle
        .BorderColor = RGB(220, 0, 0)
    End With

    ' Kenar payı
    If pKenar > 0 Then
        Set lb = frOn.Controls.Add("Forms.Label.1")
        With lb
            .Left = sx + pKenar * sc: .Top = sy + pKenar * sc
            .Width = Max2((pMatW - 2 * pKenar) * sc, 1): .Height = Max2((pMatH - 2 * pKenar) * sc, 1)
            .BackStyle = fmBackStyleTransparent
            .BorderStyle = fmBorderStyleSingle
            .BorderColor = RGB(200, 200, 200)
        End With
    End If

    ' Nesneler
    For k = 1 To nItem
        topDolu = topDolu + iW(k) * iH(k)
        If rS(k) = gosterPlaka Then
            Konum k, x, y, w, h
            adet = adet + 1
            dolu = dolu + iW(k) * iH(k)
            If c < MAKS_ONIZLEME Then
                c = c + 1
                Set lb = frOn.Controls.Add("Forms.Label.1")
                With lb
                    .Left = sx + (pKenar + x) * sc
                    .Top = sy + (pMatH - pKenar - y - h) * sc
                    .Width = Max2(w * sc, 1)
                    .Height = Max2(h * sc, 1)
                    .BorderStyle = fmBorderStyleSingle
                    If rR(k) Then
                        .BackColor = RGB(255, 214, 153)
                        .BorderColor = RGB(190, 110, 0)
                    Else
                        .BackColor = RGB(184, 214, 245)
                        .BorderColor = RGB(30, 90, 160)
                    End If
                    .ControlTipText = "Nesne " & iSrc(k) & IIf(iKop(k) > 1, " (kopya " & iKop(k) & ")", "") & ": " & _
                                      OlcuYazi(iW(k)) & " x " & OlcuYazi(iH(k)) & " " & BirimAdi() & _
                                      IIf(rR(k), "  - 90° döndürüldü", "")
                End With
            End If
        End If
    Next k

    lblPlaka.Caption = "Plaka " & gosterPlaka & " / " & rSheets
    btnOnceki.Enabled = (gosterPlaka > 1)
    btnSonraki.Enabled = (gosterPlaka < rSheets)

    lblBilgi.Caption = "Bu plaka: " & adet & " nesne   |   Kullanım %" & Format(100 * dolu / (pMatW * pMatH), "0.0") & _
                       "   |   Fire %" & Format(100 - 100 * dolu / (pMatW * pMatH), "0.0") & vbCrLf & _
                       "Toplam: " & nItem & " nesne, " & rSheets & " plaka (" & OlcuYazi(pMatW) & " x " & _
                       OlcuYazi(pMatH) & " " & BirimAdi() & ")   |   Ortalama kullanım %" & _
                       Format(100 * topDolu / (rSheets * pMatW * pMatH), "0.0")
End Sub

' k. parçanın plaka içi konumu (kenar payı hariç, başlangıç köşesine göre) ve gerçek ölçüsü (mm)
Private Sub Konum(ByVal k As Long, ByRef x As Double, ByRef y As Double, ByRef w As Double, ByRef h As Double)
    If rR(k) Then
        w = iH(k): h = iW(k)
    Else
        w = iW(k): h = iH(k)
    End If
    x = rX(k)
    y = rY(k)
    Select Case pBasla
        Case 1  ' Sol üst
            y = pSH - y - (h + pGap)
        Case 2  ' Sağ alt
            x = pSW - x - (w + pGap)
        Case 3  ' Sağ üst
            x = pSW - x - (w + pGap)
            y = pSH - y - (h + pGap)
    End Select
End Sub

'=====================================================================
'  UYGULA (nesneleri CorelDRAW'da yerleştirir)
'=====================================================================
Private Function Uygula() As Boolean
    Dim doc As Object, sr As Object, eskiBirim As Long, grupAcik As Boolean, msg As String
    Dim shp() As Object, sayfalar() As Object, grp() As Object, lyr As Object, frm As Object
    Dim k As Long, s As Long, yer As Long, ara As Double
    Dim ox As Double, oy As Double, bx As Double, byy As Double, bw As Double, bh As Double
    Dim x As Double, y As Double, w As Double, h As Double
    Dim dolu() As Double, adetPl() As Long

    Hesapla
    gosterPlaka = 1
    Ciz
    If Not hesapOK Then
        MsgBox hesapMesaj, vbExclamation, "Otomatik Dizme"
        Exit Function
    End If

    Set doc = AktifBelge()
    If doc Is Nothing Then
        MsgBox "Seçili nesneler okunamadı. Nesneleri tekrar seçip deneyin.", vbExclamation, "Otomatik Dizme"
        Exit Function
    End If
    yer = cmbPlakaYer.ListIndex
    ara = SayiOku(txtPlakaAra.Text) * birimCarpan

    On Error GoTo Hata
    adim = "hazırlık"
    eskiBirim = doc.Unit
    doc.Unit = cdrMillimeter
    doc.BeginCommandGroup "Otomatik Dizme"
    grupAcik = True
    Application.Optimization = True

    ' Sayfalar
    ReDim sayfalar(1 To rSheets)
    Set sayfalar(1) = doc.ActivePage
    If chkSayfaBoyut.Value Then sayfalar(1).SetSize pMatW, pMatH
    If yer = 2 Then
        For s = 2 To rSheets
            Set sayfalar(s) = doc.AddPages(1)
            sayfalar(s).SetSize sayfalar(1).SizeWidth, sayfalar(1).SizeHeight
        Next s
    End If

    ' Kopyaları oluştur (orijinaller henüz yerindeyken)
    adim = "kopyalar"
    ReDim shp(1 To nItem)
    For k = 1 To nItem
        If iKop(k) = 1 Then
            Set shp(k) = secS(iSrc(k))
        Else
            Set shp(k) = secS(iSrc(k)).Duplicate
        End If
    Next k

    ' Grup listeleri
    ReDim grp(1 To rSheets)
    For s = 1 To rSheets
        Set grp(s) = YeniShapeRange()
    Next s
    ReDim dolu(1 To rSheets): ReDim adetPl(1 To rSheets)

    ' Yerleştir
    adim = "yerleştirme"
    For k = 1 To nItem
        s = rS(k)
        If yer = 2 And s > 1 Then shp(k).MoveToLayer sayfalar(s).ActiveLayer
        If rR(k) Then shp(k).Rotate 90
        shp(k).GetBoundingBox bx, byy, bw, bh, pKontur
        PlakaOrijin s, yer, ara, sayfalar, ox, oy
        Konum k, x, y, w, h
        shp(k).Move (ox + pKenar + x) - bx, (oy + pKenar + y) - byy
        GrupEkle grp(s), shp(k)
        dolu(s) = dolu(s) + iW(k) * iH(k)
        adetPl(s) = adetPl(s) + 1
    Next k

    ' Çerçeve, etiket, grup
    adim = "çerçeve / etiket"
    For s = 1 To rSheets
        PlakaOrijin s, yer, ara, sayfalar, ox, oy
        If yer = 2 Then
            Set lyr = sayfalar(s).ActiveLayer
        Else
            Set lyr = doc.ActiveLayer
        End If
        If chkCerceve.Value Then
            Set frm = lyr.CreateRectangle2(ox, oy, pMatW, pMatH)
            CerceveBicimle frm, s
            GrupEkle grp(s), frm
        End If
        If chkEtiket.Value Then
            EtiketYaz lyr, ox, oy - 8, "Plaka " & s & "/" & rSheets & "  •  " & OlcuYazi(pMatW) & " x " & _
                      OlcuYazi(pMatH) & " " & BirimAdi() & "  •  " & adetPl(s) & " adet  •  Kullanım %" & _
                      Format(100 * dolu(s) / (pMatW * pMatH), "0.0"), grp(s)
        End If
        If chkGrupla.Value Then Grupla grp(s), s
    Next s

    Application.Optimization = False
    doc.EndCommandGroup
    grupAcik = False
    doc.Unit = eskiBirim
    Application.ActiveWindow.Refresh

    lblBilgi.Caption = "Tamamlandı: " & nItem & " nesne " & rSheets & " plakaya dizildi." & vbCrLf & _
                       "Beğenmezseniz CorelDRAW'da Ctrl+Z ile tek adımda geri alabilirsiniz."
    Uygula = True
    Exit Function

Hata:
    msg = Err.Description & " [" & Err.Number & "]"
    On Error Resume Next
    Application.Optimization = False
    If grupAcik Then doc.EndCommandGroup
    doc.Unit = eskiBirim
    Application.ActiveWindow.Refresh
    MsgBox "Dizme sırasında hata oluştu (" & adim & "):" & vbCrLf & msg & vbCrLf & vbCrLf & _
           "Yapılan değişiklikleri Ctrl+Z ile geri alabilirsiniz.", vbExclamation, "Otomatik Dizme"
    Uygula = False
End Function

' Etkin belge (yoksa Nothing)
Private Function AktifBelge() As Object
    On Error Resume Next
    If Application.Documents.Count = 0 Then Exit Function
    Set AktifBelge = Application.ActiveDocument
End Function

' Belgedeki seçili nesneler (ShapeRange). Okunamazsa Nothing.
Private Function SecimAl(doc As Object) As Object
    Dim sr As Object
    On Error Resume Next
    If doc Is Nothing Then Exit Function
    Set sr = doc.SelectionRange
    If sr Is Nothing Then Set sr = Application.ActiveSelectionRange
    If sr Is Nothing Then Set sr = doc.Selection.Shapes.All
    Set SecimAl = sr
End Function

' Seçimdeki nesneleri secS dizisine alır, sayısını döndürür
Private Function SecimDiziyeAl(sr As Object) As Long
    Dim o As Object, m As Long
    Erase secS
    If sr Is Nothing Then Exit Function
    For Each o In sr
        m = m + 1
        ReDim Preserve secS(1 To m)
        Set secS(m) = o
    Next o
    SecimDiziyeAl = m
End Function

' Plakanın sol-alt köşesi (mm). Plakalar sayfanın sol üstünden başlar.
Private Sub PlakaOrijin(ByVal s As Long, ByVal yer As Long, ByVal ara As Double, sayfalar() As Object, _
                        ByRef ox As Double, ByRef oy As Double)
    Dim pr As Object
    If yer = 2 Then
        Set pr = sayfalar(s).BoundingBox
    Else
        Set pr = sayfalar(1).BoundingBox
    End If
    ox = pr.x
    oy = pr.y + pr.Height - pMatH
    Select Case yer
        Case 0: ox = ox + (s - 1) * (pMatW + ara)
        Case 1: oy = oy - (s - 1) * (pMatH + ara)
    End Select
End Sub

Private Sub CerceveBicimle(frm As Object, ByVal s As Long)
    On Error Resume Next
    frm.Fill.ApplyNoFill
    frm.Outline.Width = 0.25
    frm.Outline.Color.RGBAssign 255, 0, 0
    frm.Name = "Plaka " & s
    frm.OrderToBack
End Sub

Private Sub EtiketYaz(lyr As Object, ByVal x As Double, ByVal y As Double, ByVal yazi As String, grp As Object)
    Dim t As Object
    On Error Resume Next
    Set t = lyr.CreateArtisticText(x, y, yazi)
    If t Is Nothing Then Exit Sub
    t.Text.Story.Size = 12
    t.Fill.UniformColor.RGBAssign 200, 0, 0
    t.Name = "Plaka bilgisi"
    GrupEkle grp, t
End Sub

Private Function YeniShapeRange() As Object
    On Error Resume Next
    Set YeniShapeRange = Application.CreateShapeRange
End Function

Private Sub GrupEkle(grp As Object, shp As Object)
    On Error Resume Next
    If Not grp Is Nothing Then grp.Add shp
End Sub

Private Sub Grupla(grp As Object, ByVal s As Long)
    Dim g As Object
    On Error Resume Next
    If grp Is Nothing Then Exit Sub
    If grp.Count > 1 Then
        Set g = grp.Group
        g.Name = "Plaka " & s & " grubu"
    End If
End Sub

'=====================================================================
'  DİZME MOTORU (MaxRects, birden çok kural ve sıralama denenir)
'=====================================================================
Private Function EnIyiYerlesim(ByVal n As Long, aw() As Double, ah() As Double, ByVal SW As Double, ByVal SH As Double, _
                               ByVal allowRot As Boolean, ByVal kalite As Long, ByVal arama As Long) As Long
    Dim ord() As Long, key() As Double, i As Long, rule As Long, d As Long, s As Long
    Dim pS() As Long, pX() As Double, pY() As Double, pR() As Boolean, ext As Double
    Dim bestS As Long, bestExt As Double, t0 As Single, rastgele As Single

    ReDim key(1 To n)
    bestS = 0
    t0 = Timer

    ' 1) Genel en iyi seçim (her adımda tüm parçalar arasından en uygunu)
    For i = 1 To n
        key(i) = Max2(aw(i), ah(i)) * 10000000000# + aw(i) * ah(i)
    Next i
    ord = SiraOlustur(key, n)
    For rule = 0 To 4
        s = PackGlobal(rule, n, aw, ah, ord, SW, SH, allowRot, pS, pX, pY, pR, ext)
        EnIyiyiTut s, ext, pS, pX, pY, pR, bestS, bestExt
    Next rule

    ' 2) Farklı sıralamalarla sıralı yerleştirme
    If kalite >= 1 Then
        For d = 1 To 5
            For i = 1 To n
                Select Case d
                    Case 1: key(i) = aw(i) * ah(i)
                    Case 2: key(i) = Max2(aw(i), ah(i))
                    Case 3: key(i) = aw(i)
                    Case 4: key(i) = ah(i)
                    Case Else: key(i) = aw(i) + ah(i)
                End Select
            Next i
            ord = SiraOlustur(key, n)
            For rule = 0 To 4
                s = PackSeq(rule, n, aw, ah, ord, SW, SH, allowRot, pS, pX, pY, pR, ext)
                EnIyiyiTut s, ext, pS, pX, pY, pR, bestS, bestExt
            Next rule
            DoEvents
        Next d
    End If

    ' 3) Karışık sıralamalarla ek arama
    If kalite >= 2 Then
        rastgele = Rnd(-1)
        Randomize 7
        For d = 1 To arama
            For i = 1 To n
                key(i) = aw(i) * ah(i) * (0.6 + 0.8 * Rnd)
            Next i
            ord = SiraOlustur(key, n)
            For rule = 0 To 4
                s = PackSeq(rule, n, aw, ah, ord, SW, SH, allowRot, pS, pX, pY, pR, ext)
                EnIyiyiTut s, ext, pS, pX, pY, pR, bestS, bestExt
            Next rule
            lblBilgi.Caption = "En iyi sonuç aranıyor... " & d & " / " & arama
            DoEvents
            If Abs(Timer - t0) > 30 Then Exit For
        Next d
    End If

    EnIyiYerlesim = bestS
End Function

Private Sub EnIyiyiTut(ByVal s As Long, ByVal ext As Double, pS() As Long, pX() As Double, pY() As Double, pR() As Boolean, _
                       ByRef bestS As Long, ByRef bestExt As Double)
    If s <= 0 Then Exit Sub
    If bestS = 0 Or s < bestS Or (s = bestS And ext < bestExt - EPS) Then
        bestS = s
        bestExt = ext
        rS = pS: rX = pX: rY = pY: rR = pR
    End If
End Sub

' Anahtara göre büyükten küçüğe sıra (kararlı ekleme sıralaması)
Private Function SiraOlustur(key() As Double, ByVal n As Long) As Long()
    Dim o() As Long, i As Long, j As Long, t As Long
    ReDim o(1 To n)
    For i = 1 To n: o(i) = i: Next i
    For i = 2 To n
        t = o(i)
        j = i - 1
        Do While j >= 1
            If key(t) > key(o(j)) + EPS Then
                o(j + 1) = o(j)
                j = j - 1
            Else
                Exit Do
            End If
        Loop
        o(j + 1) = t
    Next i
    SiraOlustur = o
End Function

Private Function PackGlobal(ByVal rule As Long, ByVal n As Long, aw() As Double, ah() As Double, ord() As Long, _
                            ByVal SW As Double, ByVal SH As Double, ByVal allowRot As Boolean, _
                            pS() As Long, pX() As Double, pY() As Double, pR() As Boolean, ByRef ext As Double) As Long
    Dim placed() As Boolean, kalan As Long, plaka As Long, plakadaki As Long
    Dim k As Long, i As Long, bi As Long, lw As Double, lh As Double
    Dim x As Double, y As Double, rt As Boolean, s1 As Double, s2 As Double
    Dim gx As Double, gy As Double, gr As Boolean, g1 As Double, g2 As Double

    ReDim placed(1 To n): ReDim pS(1 To n): ReDim pX(1 To n): ReDim pY(1 To n): ReDim pR(1 To n)
    kalan = n: plaka = 1: plakadaki = 0
    FreeInit SW, SH

    Do While kalan > 0
        bi = 0: g1 = 1E+300: g2 = 1E+300
        lw = -1: lh = -1
        For k = 1 To n
            i = ord(k)
            If Not placed(i) Then
                If Not (Abs(aw(i) - lw) < EPS And Abs(ah(i) - lh) < EPS) Then
                    lw = aw(i): lh = ah(i)
                    If EnIyiYer(rule, aw(i), ah(i), allowRot, x, y, rt, s1, s2) Then
                        If s1 < g1 - EPS Or (Abs(s1 - g1) <= EPS And s2 < g2 - EPS) Then
                            bi = i: g1 = s1: g2 = s2: gx = x: gy = y: gr = rt
                        End If
                    End If
                End If
            End If
        Next k

        If bi = 0 Then
            If plakadaki = 0 Then PackGlobal = -1: Exit Function
            plaka = plaka + 1
            plakadaki = 0
            FreeInit SW, SH
        Else
            Yerlestir bi, plaka, gx, gy, gr, aw, ah, placed, pS, pX, pY, pR
            kalan = kalan - 1
            plakadaki = plakadaki + 1
        End If
    Loop

    ext = SonPlakaAlani(n, plaka, aw, ah, pS, pX, pY, pR)
    PackGlobal = plaka
End Function

Private Function PackSeq(ByVal rule As Long, ByVal n As Long, aw() As Double, ah() As Double, ord() As Long, _
                         ByVal SW As Double, ByVal SH As Double, ByVal allowRot As Boolean, _
                         pS() As Long, pX() As Double, pY() As Double, pR() As Boolean, ByRef ext As Double) As Long
    Dim placed() As Boolean, kalan As Long, plaka As Long, plakadaki As Long
    Dim k As Long, i As Long, x As Double, y As Double, rt As Boolean, s1 As Double, s2 As Double

    ReDim placed(1 To n): ReDim pS(1 To n): ReDim pX(1 To n): ReDim pY(1 To n): ReDim pR(1 To n)
    kalan = n: plaka = 0

    Do While kalan > 0
        plaka = plaka + 1
        plakadaki = 0
        FreeInit SW, SH
        For k = 1 To n
            i = ord(k)
            If Not placed(i) Then
                If EnIyiYer(rule, aw(i), ah(i), allowRot, x, y, rt, s1, s2) Then
                    Yerlestir i, plaka, x, y, rt, aw, ah, placed, pS, pX, pY, pR
                    kalan = kalan - 1
                    plakadaki = plakadaki + 1
                End If
            End If
        Next k
        If plakadaki = 0 Then PackSeq = -1: Exit Function
    Loop

    ext = SonPlakaAlani(n, plaka, aw, ah, pS, pX, pY, pR)
    PackSeq = plaka
End Function

Private Sub Yerlestir(ByVal i As Long, ByVal plaka As Long, ByVal x As Double, ByVal y As Double, ByVal rt As Boolean, _
                      aw() As Double, ah() As Double, placed() As Boolean, _
                      pS() As Long, pX() As Double, pY() As Double, pR() As Boolean)
    placed(i) = True
    pS(i) = plaka: pX(i) = x: pY(i) = y: pR(i) = rt
    If rt Then
        PlaceRect x, y, ah(i), aw(i)
    Else
        PlaceRect x, y, aw(i), ah(i)
    End If
End Sub

' Bir parça için mevcut boş alanlar arasında en iyi yer
Private Function EnIyiYer(ByVal rule As Long, ByVal w0 As Double, ByVal h0 As Double, ByVal allowRot As Boolean, _
                          ByRef bxp As Double, ByRef byp As Double, ByRef bRot As Boolean, _
                          ByRef b1 As Double, ByRef b2 As Double) As Boolean
    Dim o As Long, r As Long, w As Double, h As Double, s1 As Double, s2 As Double
    b1 = 1E+300: b2 = 1E+300
    For o = 0 To 1
        If o = 0 Then
            w = w0: h = h0
        Else
            w = h0: h = w0
        End If
        If o = 0 Or (allowRot And Abs(w - h) > EPS) Then
            For r = 1 To fc
                If w <= fw(r) + EPS And h <= fh(r) + EPS Then
                    Puan rule, r, w, h, s1, s2
                    If s1 < b1 - EPS Or (Abs(s1 - b1) <= EPS And s2 < b2 - EPS) Then
                        b1 = s1: b2 = s2
                        bxp = fx(r): byp = fy(r): bRot = (o = 1)
                        EnIyiYer = True
                    End If
                End If
            Next r
        End If
    Next o
End Function

Private Sub Puan(ByVal rule As Long, ByVal r As Long, ByVal w As Double, ByVal h As Double, _
                 ByRef s1 As Double, ByRef s2 As Double)
    Dim lw As Double, lh As Double
    lw = fw(r) - w
    lh = fh(r) - h
    Select Case rule
        Case 0      ' En iyi kısa kenar uyumu
            s1 = Min2(lw, lh): s2 = Max2(lw, lh)
        Case 1      ' En iyi uzun kenar uyumu
            s1 = Max2(lw, lh): s2 = Min2(lw, lh)
        Case 2      ' En iyi alan uyumu
            s1 = fw(r) * fh(r) - w * h: s2 = Min2(lw, lh)
        Case 3      ' Satır satır (alttan)
            s1 = fy(r) + h: s2 = fx(r)
        Case Else   ' Sütun sütun (soldan)
            s1 = fx(r) + w: s2 = fy(r)
    End Select
End Sub

Private Function SonPlakaAlani(ByVal n As Long, ByVal plaka As Long, aw() As Double, ah() As Double, _
                               pS() As Long, pX() As Double, pY() As Double, pR() As Boolean) As Double
    Dim i As Long, w As Double, h As Double, mx As Double, my As Double
    For i = 1 To n
        If pS(i) = plaka Then
            If pR(i) Then
                w = ah(i): h = aw(i)
            Else
                w = aw(i): h = ah(i)
            End If
            If pX(i) + w > mx Then mx = pX(i) + w
            If pY(i) + h > my Then my = pY(i) + h
        End If
    Next i
    SonPlakaAlani = mx * my
End Function

Private Sub FreeInit(ByVal W As Double, ByVal H As Double)
    ReDim fx(1 To 1): ReDim fy(1 To 1): ReDim fw(1 To 1): ReDim fh(1 To 1)
    fx(1) = 0: fy(1) = 0: fw(1) = W: fh(1) = H
    fc = 1
End Sub

Private Sub TAdd(ByVal x As Double, ByVal y As Double, ByVal w As Double, ByVal h As Double)
    If w <= EPS Or h <= EPS Then Exit Sub
    tc = tc + 1
    tx(tc) = x: ty(tc) = y: tw(tc) = w: th(tc) = h
End Sub

Private Function TIn(ByVal i As Long, ByVal j As Long) As Boolean
    TIn = (tx(i) >= tx(j) - EPS And ty(i) >= ty(j) - EPS And _
           tx(i) + tw(i) <= tx(j) + tw(j) + EPS And ty(i) + th(i) <= ty(j) + th(j) + EPS)
End Function

Private Sub PlaceRect(ByVal x As Double, ByVal y As Double, ByVal w As Double, ByVal h As Double)
    Dim i As Long, j As Long, cnt As Long, keep() As Boolean
    Dim ax As Double, ay As Double, aww As Double, ahh As Double

    tc = 0
    ReDim tx(1 To fc * 4 + 4): ReDim ty(1 To fc * 4 + 4)
    ReDim tw(1 To fc * 4 + 4): ReDim th(1 To fc * 4 + 4)

    For i = 1 To fc
        ax = fx(i): ay = fy(i): aww = fw(i): ahh = fh(i)
        If x >= ax + aww - EPS Or x + w <= ax + EPS Or y >= ay + ahh - EPS Or y + h <= ay + EPS Then
            TAdd ax, ay, aww, ahh
        Else
            If x > ax + EPS Then TAdd ax, ay, x - ax, ahh
            If x + w < ax + aww - EPS Then TAdd x + w, ay, ax + aww - x - w, ahh
            If y > ay + EPS Then TAdd ax, ay, aww, y - ay
            If y + h < ay + ahh - EPS Then TAdd ax, y + h, aww, ay + ahh - y - h
        End If
    Next i

    If tc = 0 Then fc = 0: Exit Sub

    ReDim keep(1 To tc)
    For i = 1 To tc: keep(i) = True: Next i
    For i = 1 To tc
        For j = 1 To tc
            If i <> j And keep(j) Then
                If TIn(i, j) Then
                    If (Not TIn(j, i)) Or j < i Then keep(i) = False: Exit For
                End If
            End If
        Next j
    Next i

    ReDim fx(1 To tc): ReDim fy(1 To tc): ReDim fw(1 To tc): ReDim fh(1 To tc)
    cnt = 0
    For i = 1 To tc
        If keep(i) Then
            cnt = cnt + 1
            fx(cnt) = tx(i): fy(cnt) = ty(i): fw(cnt) = tw(i): fh(cnt) = th(i)
        End If
    Next i
    fc = cnt
End Sub

Private Function Min2(ByVal a As Double, ByVal b As Double) As Double
    If a < b Then Min2 = a Else Min2 = b
End Function

Private Function Max2(ByVal a As Double, ByVal b As Double) As Double
    If a > b Then Max2 = a Else Max2 = b
End Function
